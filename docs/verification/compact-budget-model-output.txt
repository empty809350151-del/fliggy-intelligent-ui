root = Col([intro, tool, note])
$people = 2
$days = 3
$budget = 6000
$hotel = 700
$food = 180
$transport = 800
$other = 600
$details = false
people = @Clamp(@Default($people, 1), 1, 30)
days = @Clamp(@Default($days, 1), 1, 60)
budget = @Clamp(@Default($budget, 0), 0, 10000000)
hotel = @Clamp(@Default($hotel, 0), 0, 1000000)
food = @Clamp(@Default($food, 0), 0, 100000)
transport = @Clamp(@Default($transport, 0), 0, 1000000)
other = @Clamp(@Default($other, 0), 0, 1000000)
total = hotel * days + food * days * people + transport + other
perPerson = @Round(total / people, 2)
over = total > budget
intro = Text("先把人数、天数和总预算定下来，下方金额可编辑；展开“调整费用明细”可修改住宿、餐饮等分项。")
controls = Row([Input($people, "人数", "number"), Input($days, "天数", "number"), Input($budget, "总预算（元）", "number")], 2, "stretch", null, true)
summary = Col([Row([Caption("预计总额"), Spacer(), Text("¥" + @Format(total, "number", 2), "md", "ink", "semibold", false, false, true)], 2), Row([Caption("人均"), Spacer(), Text("¥" + @Format(perPerson, "number", 2), "sm", "ink", "medium", false, false, true)], 2), Badge(over ? "超支 ¥" + @Format(total - budget, "number", 2) : "预算剩余 ¥" + @Format(budget - total, "number", 2), over ? "danger" : "success")], 2)
toggle = Button($details ? "收起费用明细" : "调整费用明细", @Set($details, !$details), "ghost", null, "sm", $details ? "chevron-up" : "sliders-horizontal")
detailInputs = Col([Input($hotel, "住宿每晚（整团）", "number"), Input($food, "餐饮每人每天", "number"), Input($transport, "交通总计（整团）", "number"), Input($other, "门票及其他（整团）", "number")], 3)
copyText = "旅行分账：" + @Plural(people, "人") + "，" + @Plural(days, "天") + "；预计总额 ¥" + @Format(total, "number", 2) + "，人均 ¥" + @Format(perPerson, "number", 2) + (over ? "，超支 ¥" + @Format(total - budget, "number", 2) : "，预算剩余 ¥" + @Format(budget - total, "number", 2))
copyBtn = Button("复制分账结果", @CopyToClipboard(copyText), "outline", null, "sm", "copy")
tool = TravelModule([controls, summary, toggle, $details ? detailInputs : null], "budget", "旅行预算与分账", "人数、天数可调，自动标出人均与超支", [copyBtn])
note = Text(over ? "当前估算已超支。建议优先压缩住宿或餐饮，或用“复制分账结果”发给同伴一起核对；这里不会自动下单。" : "当前在预算内。建议留出应急和购物预算，不要把剩余金额全部用于升级住宿。")
